def sha256_file(path, block_size=8 * 1024 * 1024):
    h = hashlib.sha256()
    with open(path, 'rb') as f:
        for block in iter(lambda: f.read(block_size), b''):
            h.update(block)
    return h.hexdigest()


def write_csv(df, filename):

def norm_code(value):
    s = str(value if value is not None else '').strip()
    if s.endswith('.0') and s[:-2].isdigit():
        s = s[:-2]
    return s.upper()


def norm_market(value):
    s = str(value if value is not None else '').strip()
    u = s.upper()
    if '上市' in s or u in {'TWSE', 'TSE', 'LISTED', '1'}:
        return '上市'
    if '上櫃' in s or u in {'TPEX', 'OTC', 'TPEx'.upper(), '2'}:
        return '上櫃'
    return s


def parse_num(value):
    if value is None:
        return None
    s = str(value).strip().replace(',', '').replace(' ', '')
    if not s or s in {'---', '--', '-', 'N/A', 'NA', 'None', 'nan'}:
        return None
    s = re.sub(r'<[^>]*>', '', s)
    try:
        return float(s)
    except Exception:
        return None


def pick_col(columns, aliases):
    exact = {str(c).strip(): c for c in columns}
    for a in aliases:
        if a in exact:
            return exact[a]
    lower = {str(c).strip().lower(): c for c in columns}
    for a in aliases:
        if a.lower() in lower:
            return lower[a.lower()]
    return None

                sheets_scanned = 0

                        continue

                    header_text = [

                    ]

def first_value(d, keys):
    for k in keys:
        if k in d:
            return d.get(k)
    return None


def walk_official_rows(obj, out=None):
    if out is None:
        out = []

    if isinstance(obj, dict):
        # TWSE MI_INDEX tables use fields + data arrays.
        fields = obj.get('fields')
        data = obj.get('data')

        if isinstance(fields, list) and isinstance(data, list):
            for item in data:
                if isinstance(item, list):
                    out.append({str(k): v for k, v in zip(fields, item)})
                elif isinstance(item, dict):
                    out.append(item)

        if any(k in obj for k in CODE_KEYS):
            out.append(obj)

        for k, v in obj.items():
            if k not in {'fields', 'data'} and isinstance(v, (dict, list)):
                walk_official_rows(v, out)

    elif isinstance(obj, list):
        for item in obj:
            walk_official_rows(item, out)

    return out


def load_official_map(path, market):
    raw_map = {}
    if not path.exists():
        return raw_map

    with open(path, 'r', encoding='utf-8-sig') as f:
        payload = json.load(f)

    candidates = walk_official_rows(payload)
    for rec in candidates:
        code = norm_code(first_value(rec, CODE_KEYS))
        if not code:
            continue
        # Some recursive walks can discover the same table row more than once.
        raw_map[code] = rec

    return raw_map

def github_request(method, url, token, **kwargs):
    headers = kwargs.pop('headers', {})
    headers.update({
        'Authorization': f'Bearer {token}',
        'Accept': 'application/vnd.github+json',
        'X-GitHub-Api-Version': '2022-11-28'
    })

    response = requests.request(
        method, url,
        headers=headers,
        timeout=90,
        **kwargs
    )

    if response.status_code >= 400:
        raise RuntimeError(
            f'GitHub API {method} {url} failed: '
            f'{response.status_code} {response.text[:1200]}'
        )

    if response.text:
        try:
            return response.json()
        except Exception:
            return {'text': response.text[:2000]}

    return {}

                    'path': rel,
