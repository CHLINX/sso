def log(message):

def sha256_file(path):
    h = hashlib.sha256()

    with open(path, "rb") as f:
        for block in iter(
            lambda: f.read(1024 * 1024),
            b""
        ):
            h.update(block)

    return h.hexdigest()


def sha256_bytes(data):
    return hashlib.sha256(data).hexdigest()


def decode_bytes(data):
    for encoding in (
        "utf-8-sig",
        "utf-8",
        "cp950",
        "big5",
        "latin-1",
    ):
        try:
            return data.decode(encoding)
        except Exception:
            pass

    return data.decode(
        "utf-8",
        errors="replace"
    )

def github_request(method, url, token, **kwargs):
    import requests

    headers = kwargs.pop("headers", {})

    headers.update({
        "Accept": "application/vnd.github+json",
        "X-GitHub-Api-Version": "2022-11-28",
    })

    if token:
        headers["Authorization"] = f"Bearer {token}"

    headers.setdefault(
        "User-Agent",
        "SSO-v1.3.7-lineage-audit"
    )

    response = requests.request(
        method,
        url,
        headers=headers,
        timeout=90,
        **kwargs
    )

    if response.status_code >= 400:
        raise RuntimeError(
            f"GitHub {method} {url} "
            f"HTTP {response.status_code}: "
            f"{response.text[:1200]}"
        )

    return response

def walk_files(root, limit=60000):
    results = []

    for base, dirs, files in os.walk(root):
        dirs[:] = [
            d for d in dirs
            if d not in SKIP_DIR_NAMES
            and not d.startswith(".")
        ]

        for name in files:
            path = Path(base) / name

            try:
                if path.is_symlink():
                    continue

                stat = path.stat()

                results.append({
                    "path": path,
                    "size": stat.st_size,
                    "mtime": stat.st_mtime,
                })

            except Exception:
                pass

            if len(results) >= limit:
                return results

    return results


def read_text_file(path, max_bytes=15 * 1024 * 1024):
    path = Path(path)

    try:
        if path.stat().st_size > max_bytes:
            return None

        raw = path.read_bytes()

        if path.name.lower().endswith(".gz"):
            raw = gzip.decompress(raw)

        return decode_bytes(raw)

    except Exception:
        return None


def read_zip_member(zip_file, member_name, max_bytes=15 * 1024 * 1024):
    try:
        info = zip_file.getinfo(member_name)

        if info.file_size > max_bytes:
            return None

        return decode_bytes(
            zip_file.read(member_name)
        )

    except Exception:
        return None


def read_xlsx_rows(path):
    try:
        import openpyxl

        workbook = openpyxl.load_workbook(
            path,
            read_only=True,
            data_only=True
        )

        results = []

        for worksheet in workbook.worksheets:
            iterator = worksheet.iter_rows(
                values_only=True
            )

            first_row = next(iterator, None)

            if not first_row:
                continue

            headers = [
                str(value).strip()
                if value is not None
                else f"欄{i + 1}"
                for i, value in enumerate(first_row)
            ]

            for row_number, values in enumerate(
                iterator,
                start=2
            ):
                row = {
                    headers[i]: (
                        values[i]
                        if i < len(values)
                        else None
                    )
                    for i in range(len(headers))
                }

                results.append((
                    worksheet.title,
                    row_number,
                    row
                ))

        workbook.close()

        return results

    except Exception:
        return []


def parse_csv_text(text):
    if text is None:
        return [], []

    try:
        sample = text[:8192]

        try:
            dialect = csv.Sniffer().sniff(
                sample,
                delimiters=",\t;|"
            )
        except Exception:
            dialect = csv.excel

        rows = list(
            csv.DictReader(
                io.StringIO(text),
                dialect=dialect
            )
        )

        headers = (
            list(rows[0].keys())
            if rows
            else []
        )

        return headers, rows

    except Exception:
        return [], []


def has_target_date(row):
    for key, value in row.items():
        if re.search(
            r"日期|date|交易日",
            str(key),
            re.I
        ):
            text = str(value).strip()
            digits = re.sub(r"\D", "", text)

            # 同時辨識西元及民國日期
            if (
                "20261109" in digits
                or "1151109" in digits
            ):
                return True

    return False


def norm_ticker(value):
    text = str(value or "").strip()

    match = re.fullmatch(
        r"(?:股票代號[:：]?\s*)?(\d{4,6})(?:\.0)?",
        text
    )

    return match.group(1) if match else None


def find_ticker_in_row(row):
    # 優先從股票代號欄位辨識
    for key, value in row.items():
        if any(
            word in str(key).lower()
            for word in (
                "代號",
                "證券代號",
                "stock",
                "ticker",
                "symbol",
                "code",
            )
        ):
            ticker = norm_ticker(value)

            if ticker in TARGETS:
                return ticker

    # 再檢查其他欄位
    for value in row.values():
        ticker = norm_ticker(value)

        if ticker in TARGETS:
            return ticker

    return None


def json_walk(obj, prefix="$"):
    if isinstance(obj, dict):
        yield prefix, obj

        for key, value in obj.items():
            yield from json_walk(
                value,
                f"{prefix}.{key}"
            )

    elif isinstance(obj, list):
        for index, value in enumerate(
            obj[:200000]
        ):
            yield from json_walk(
                value,
                f"{prefix}[{index}]"
            )

def extract_refs_from_text(text, origin):
    refs = []

    if not text:
        return refs

    hashes = set(
        re.findall(
            r"(?i)\b[a-f0-9]{64}\b",
            text
        )
    )

    for sha in hashes:
        refs.append({
            "kind": "sha256",
            "value": sha.lower(),
            "origin": origin,
        })

    patterns = re.findall(
        r"""[^\r\n\t"'<>|;，；]{1,260}?"""
        r"""\.(?:csv|tsv|xlsx|xlsm|json|jsonl|json\.gz|zip|py|ipynb|gs|sql|txt|md|parquet)""",
        text,
        re.I
    )

    for value in patterns:
        value = value.strip(" ,()[]{}:：")

        if value and len(value) < 300:
            refs.append({
                "kind": "filename",
                "value": value,
                "origin": origin,
            })

    return refs


def extract_code_refs(text, origin):
    results = []

    if not text:
        return results

    patterns = re.findall(
        r"""[^\r\n\t"'<>|;，；]{1,260}?"""
        r"""\.(?:py|ipynb|gs|sql)""",
        text,
        re.I
    )

    for value in patterns:
        value = value.strip(" ,()[]{}:：")

        if value:
            results.append({
                "kind": "program",
                "value": value,
                "origin": origin,
            })

    return results

def fetch_previous_audit_zips(token, workspace):
    import requests

    downloads = []
    errors = []

    for branch in PRIOR_BRANCHES:
        try:
            branch_api = (
                f"https://api.github.com/repos/{REPO}"
                f"/branches/{quote(branch, safe='')}"
            )

            branch_data = github_request(
                "GET",
                branch_api,
                token
            ).json()

            tree_sha = (
                branch_data["commit"]["commit"]["tree"]["sha"]
            )

            tree_api = (
                f"https://api.github.com/repos/{REPO}"
                f"/git/trees/{tree_sha}?recursive=1"
            )

            tree_data = github_request(
                "GET",
                tree_api,
                token
            ).json()

            tree_items = tree_data.get("tree", [])

            zip_paths = [
                item["path"]
                for item in tree_items
                if (
                    item.get("type") == "blob"
                    and item["path"].lower().endswith(".zip")
                    and (
                        "v1.3.5" in item["path"].lower()
                        or "v1335" in item["path"].lower()
                        or "v1.3.6" in item["path"].lower()
                        or "v1336" in item["path"].lower()
                    )
                )
            ]

            if not zip_paths:
                errors.append(
                    f"{branch}: 找不到 v1.3.5/v1.3.6 ZIP"
                )
                continue

            zip_paths.sort(
                key=lambda value: (len(value), value)
            )

            remote_path = zip_paths[0]

            raw_url = (
                f"https://raw.githubusercontent.com/{REPO}/"
                f"{branch}/{quote(remote_path, safe='/')}"
            )

            response = requests.get(
                raw_url,
                headers={"User-Agent": "SSO-v1.3.7"},
                timeout=120
            )

            response.raise_for_status()

            local_path = (
                workspace
                / (branch.split("-v")[-1] + ".zip")
            )

            local_path.write_bytes(response.content)

            downloads.append({
                "branch": branch,
                "path": remote_path,
                "local": local_path,
                "sha256": sha256_bytes(response.content),
                "bytes": len(response.content),
            })

            log(
                f"[PASS] 取得前次稽核包：{branch} / "
                f"{remote_path} "
                f"({len(response.content):,} bytes)"
            )

        except Exception as exc:
            errors.append(
                f"{branch}: {type(exc).__name__}: {exc}"
            )

            log(
                f"[WARN] 前次稽核包讀取失敗："
                f"{branch} / {exc}"
            )

    return downloads, errors

    candidate_programs = []
    candidate_sources = []

    scan_code_files = {}

    # 先找被檔名直接引用的程式
    for item in project_files:
        path = item["path"]

        if (
            path.suffix.lower() in CODE_EXTS
            and (
                path.name.lower() in referenced_names
                or str(path) in candidate_files
            )
        ):
            scan_code_files[str(path)] = item

    # 再搜尋其他程式中的五檔代號、數值及目標 CSV 名稱
    code_scan_count = 0
    code_scan_bytes = 0

    for item in project_files:
        path = item["path"]

        if (
            path.suffix.lower() not in CODE_EXTS
            or str(path) in scan_code_files
        ):
            continue

        if (
            item["size"] > 8 * 1024 * 1024
            or code_scan_count >= 12000
            or code_scan_bytes > 700 * 1024 * 1024
        ):
            continue

        text = read_text_file(
            path,
            max_bytes=8 * 1024 * 1024
        )

        if text is None:
            continue

        code_scan_count += 1
        code_scan_bytes += item["size"]

        matched_names = [
            name
            for name in TARGET_NAMES
            if name in text
        ]

        ticker_matched = [
            ticker
            for ticker in TARGETS
            if ticker in text
        ]

        value_matched = [
            str(value)
            for value in TARGETS.values()
            if re.search(
                rf"(?<!\d){value}(?!\d)",
                text
            )
        ]

        if (
            matched_names
            or len(ticker_matched) >= 2
            or (
                len(value_matched) >= 3
                and len(ticker_matched) >= 1
            )
        ):
            scan_code_files[str(path)] = item

    # 擷取命中程式上下文
    for path_text, item in scan_code_files.items():
        path = Path(path_text)

        text = read_text_file(
            path,
            max_bytes=15 * 1024 * 1024
        )

        if text is None:
            continue

        lines = text.splitlines()

        needles = (
            TARGET_NAMES
            + list(TARGETS.keys())
            + [
                str(value)
                for value in TARGETS.values()
            ]
        )

        line_ids = [
            index
            for index, line in enumerate(lines)
            if any(needle in line for needle in needles)
        ]

        keep = set()

        for index in line_ids:
            for nearby in range(
                max(0, index - 3),
                min(len(lines), index + 4)
            ):
                keep.add(nearby)

        context_lines = [
            f"{index + 1}: {lines[index][:1200]}"
            for index in sorted(keep)
        ]

        if context_lines:
            code_hits.append({
                "path": path_text,
                "relative_path": relpath(path),
                "sha256": (
                    sha256_file(path)
                    if path.stat().st_size <= 100 * 1024 * 1024
                    else ""
                ),
                "hit_line_count": len(line_ids),
                "context": "\n".join(context_lines[:1200]),
            })

        if path.suffix.lower() in CODE_EXTS:
            candidate_programs.append({
                "path": path_text,
                "relative_path": relpath(path),
                "size": item["size"],
                "sha256": (
                    sha256_file(path)
                    if item["size"] <= 100 * 1024 * 1024
                    else ""
                ),
                "referenced_by_name": (
                    path.name.lower() in code_names
                    or path.name.lower() in referenced_names
                ),
                "keyword_hits": len(line_ids),
            })

    # --------------------------------------------------------
    # 追查候選來源資料列
    # 排除已知衍生報表，避免把報表自身當成上游證據
    # --------------------------------------------------------

    data_paths = {}

    for path_text, item in candidate_files.items():
        path = Path(path_text)

        if path.name in TARGET_NAMES:
            continue

        # 僅採用前次稽核明確引用的檔案
        if path.name.lower() not in referenced_names:

        if (

                ".tsv",
                ".xlsx",
                ".xlsm",

            or path.name.lower().endswith(".json.gz")
            or path.suffix.lower() in (

                ".jsonl",

    for path_text, item in data_paths.items():
        path = Path(path_text)

        if path.suffix.lower() in (".xlsx", ".xlsm"):

                ticker = find_ticker_in_row(row)

                if not ticker:
                    continue

                for key, value in row.items():
                    if not re.search(
                        r"成交股數|成交量|量|volume|trade.?volume",
                        str(key),
                        re.I
                    ):
                        continue

                    try:
                        number = float(
                            str(value).replace(",", "").strip()
                        )
                    except Exception:
                        continue

                    if number == TARGETS[ticker]:
                        source_value_hits.append({
                            "file": path_text,
                            "sheet": sheet,
                            "line": line_number,
                            "ticker": ticker,
                            "target_date_match": has_target_date(row),
                            "matched_volume_column": key,
                            "matched_volume": value,
                            "date_fields": json.dumps(
                                {
                                    k: v
                                    for k, v in row.items()
                                    if re.search(
                                        r"日期|date|交易日",
                                        str(k),
                                        re.I
                                    )
                                },
                                ensure_ascii=False,
                                default=str
                            ),
                            "row": json.dumps(
                                row,
                                ensure_ascii=False,
                                default=str
                            ),
                        })

            continue

        text = read_text_file(
            path,
            max_bytes=30 * 1024 * 1024
        )

        if text is None:
            continue

        if path.suffix.lower() in (".csv", ".tsv"):
            _, rows = parse_csv_text(text)

            for line_number, row in enumerate(
                rows,
                start=2
            ):
                ticker = find_ticker_in_row(row)

                if not ticker:
                    continue

                volume_values = []

                for key, value in row.items():
                    if re.search(
                        r"成交股數|成交量|量|volume|trade.?volume",
                        str(key),
                        re.I
                    ):
                        volume_values.append((key, str(value)))

                for key, value in volume_values:
                    try:
                        number = float(
                            value.replace(",", "").strip()
                        )
                    except Exception:
                        continue

                    if number == TARGETS[ticker]:
                        source_value_hits.append({
                            "file": path_text,
                            "line": line_number,
                            "ticker": ticker,
                            "target_date_match": has_target_date(row),
                            "matched_volume_column": key,
                            "matched_volume": value,
                            "date_fields": json.dumps(
                                {
                                    k: v
                                    for k, v in row.items()
                                    if re.search(
                                        r"日期|date|交易日",
                                        str(k),
                                        re.I
                                    )
                                },
                                ensure_ascii=False
                            ),
                            "row": json.dumps(
                                row,
                                ensure_ascii=False
                            ),
                        })

        elif (
            path.suffix.lower() in (".json", ".jsonl")
            or path.name.lower().endswith(".json.gz")
        ):
            try:
                obj = json.loads(text)

                for json_path, node in json_walk(obj):
                    if not isinstance(node, dict):
                        continue

                    ticker = find_ticker_in_row(node)

                    if not ticker:
                        continue

                    for key, value in node.items():
                        if not re.search(
                            r"成交股數|成交量|量|volume|trade.?volume",
                            str(key),
                            re.I
                        ):
                            continue

                        try:
                            number = float(
                                str(value).replace(",", "").strip()
                            )
                        except Exception:
                            continue

                        if number == TARGETS[ticker]:
                            source_value_hits.append({
                                "file": path_text,
                                "json_path": json_path,
                                "ticker": ticker,
                                "target_date_match": has_target_date(node),
                                "matched_volume_column": key,
                                "matched_volume": value,
                                "row": json.dumps(
                                    node,
                                    ensure_ascii=False
                                ),
                            })

            except Exception:
                pass

        fields = []

        for row in rows:

            path,
            "w",
            encoding="utf-8-sig",
            newline=""
        ) as f:
            writer = csv.DictWriter(
                f,
                fieldnames=fields,
                extrasaction="ignore"
            )

            writer.writeheader()
            writer.writerows(rows)

        "evidence_zip_path": str(zip_path),
        "evidence_zip_sha256": zip_sha,
        "note": (
            "此確認檔為上傳後產生的獨立紀錄；"
            "證據 ZIP 在上傳前封存，"
            "ZIP 內容與 GitHub 已核對版本一致。"
        ),
    }

    confirmation_path = (
        work / "GITHUB_上傳與main核對結果.json"
    )

    confirmation_path.write_text(
        json.dumps(
            confirmation,
            ensure_ascii=False,
            indent=2
        ),
        encoding="utf-8"
    )

    if upload_result.get("success") and branch and token:
        try:
            remote_path = (
                "audit/sso-market-data-continuation-v1/"
                f"v1.3.7_五檔正成交量上游生成追溯_{stamp}/"
                "GITHUB_上傳與main核對結果.json"
            )

            import base64

            payload = {
                "message": (
                    "SSO v1.3.7 audit: "
                    "GitHub upload and main verification"
                ),
                "content": base64.b64encode(
                    confirmation_path.read_bytes()
                ).decode(),
                "branch": branch,
            }

            github_request(
                "PUT",
                (
                    f"https://api.github.com/repos/{REPO}"
                    f"/contents/{quote(remote_path, safe='/')}"
                ),
                token,
                json=payload
            )

            raw_url = (
                f"https://raw.githubusercontent.com/{REPO}/"
                f"{branch}/{quote(remote_path, safe='/')}"
            )

            response = github_request(
                "GET",
                raw_url,
                token
            )

            if (
                sha256_bytes(response.content)
                != sha256_file(confirmation_path)
            ):
                raise RuntimeError(
                    "上傳確認檔 readback SHA256 不符"
                )

            log(
                "[GITHUB PASS] 上傳確認 sidecar readback=PASS"
            )

        except Exception as exc:
            log(
                f"[GITHUB WARN] 上傳確認 sidecar 未完成：{exc}"
            )

    # --------------------------------------------------------
    # 最終輸出
    # --------------------------------------------------------

    log("=" * 100)

    log(f"[FINAL] status={status}")

    log(
        f"[FINAL] v123 rows={target_count}; "
        f"secondary rows={secondary_count}; "
        f"source-value hits={value_count}"
    )

    log(
        f"[FINAL] candidate programs={len(candidate_programs)}; "
        f"code contexts={len(code_hits)}; "
        f"parse errors={len(parse_errors)}"
    )

    log(
        f"[FINAL] GitHub upload={upload_result['success']}; "
        f"main unchanged={main_unchanged}; "
        f"main before={main_sha_before}; "
        f"after={main_sha_after}"
    )

    log(f"[FINAL] ZIP={zip_path} SHA256={zip_sha}")
